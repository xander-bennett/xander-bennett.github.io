-- account health tier, refreshed hourly
WITH active_accounts AS (
  SELECT
    account_id,
    arr_usd,
    days_idle,
    open_opps
  FROM {{ ref('int_accounts__activity') }}
  WHERE is_active
)
SELECT
  account_id,
  arr_usd,
  CASE
    WHEN days_idle > 60 THEN 'at_risk'
    WHEN days_idle BETWEEN 30 AND 60 THEN 'attention'
    ELSE 'healthy'
  END AS health_tier
FROM active_accounts
